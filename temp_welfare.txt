  static postWelfarePayment(
    db: AppDatabaseState,
    params: {
      fundType: 'WELFARE' | 'EMERGENCY' | 'RESERVE';
      amount: number;
      beneficiary: string;
      reason: string;
      memberId?: string;
      approvedByPresident?: boolean;
      approvedBySecretary?: boolean;
      approvedByTreasurer?: boolean;
      resolutionNo?: string;
      remarks?: string;
    }
  ): { success: boolean; message: string; updatedDb?: AppDatabaseState } {
    if (params.amount <= 0) return { success: false, message: 'অনুদান/ব্যয়ের পরিমাণ সঠিক নয়!' };

    // Check emergency requirements
    if (params.fundType === 'EMERGENCY' && db.settings.requireThreeSignaturesForEmergency) {
      if (!params.approvedByPresident || !params.approvedBySecretary || !params.approvedByTreasurer) {
        return {
          success: false,
          message: 'জরুরী তহবিল অনুদানের ক্ষেত্রে সভাপতি, সাধারণ সম্পাদক ও কোষাধ্যক্ষ—তিনজনেরই অনুমোদন আবশ্যক!'
        };
      }
    }

    const cash = this.getCashBalance(db.cashTransactions);
    if (cash < params.amount) {
      return { success: false, message: `পর্যাপ্ত নগদ ক্যাশ নেই! বর্তমান ক্যাশ: ৳${cash}` };
    }

    const dateStr = new Date().toISOString().split('T')[0];
    const voucherNo = this.generateVoucherNo(db, 'WLF');

    const member = params.memberId ? (db.members || []).find(m => m.memberId === params.memberId) : undefined;

    const welfareTx: any = {
      fundId: `WLF-${Date.now()}`,
      date: dateStr,
      fundType: params.fundType,
      income: 0,
      expense: params.amount,
      beneficiary: params.beneficiary,
      memberId: member?.memberId,
      memberName: member?.fullName,
      reason: params.reason,
      amount: params.amount,
      approvedByPresident: params.approvedByPresident,
      approvedBySecretary: params.approvedBySecretary,
      approvedByTreasurer: params.approvedByTreasurer,
      approvalStatus: 'APPROVED',
      resolutionNo: params.resolutionNo,
      voucherNo,
      remarks: params.remarks,
      createdAt: new Date().toISOString()
    };

    const currentCash = this.getCashBalance(db.cashTransactions);
    const cashEntry: any = {
      transactionId: `CSH-${Date.now()}`,
      date: dateStr,
      voucherNo,
      reference: `${params.fundType} তহবিল অনুদান`,
      description: `${params.beneficiary} কে সহায়তা (${params.reason})`,
      accountId: '5020',
      accountName: 'কল্যাণ অনুদান ব্যয়',
      cashIn: 0,
      cashOut: params.amount,
      balance: currentCash - params.amount,
      sourceType: 'WELFARE',
      sourceId: welfareTx.fundId,
      createdBy: 'কোষাধ্যক্ষ',
      createdAt: new Date().toISOString()
    };

    return {
      success: true,
      message: `${params.fundType} তহবিল অনুদান ভাউচার ${voucherNo} সফলভাবে প্রদান করা হয়েছে!`,
      updatedDb: {
        ...db,
        welfareTransactions: [welfareTx, ...(db.welfareTransactions || [])],
        cashTransactions: [...db.cashTransactions, cashEntry],
        auditLogs: [
          {
            auditId: `AUD-${Date.now()}`,
            userId: db.activeUserId || 'SYSTEM',
            userName: 'কোষাধ্যক্ষ',
            dateTime: new Date().toISOString(),
            module: 'WELFARE',
            action: 'PAYMENT',
            recordId: welfareTx.fundId,
            newValue: `${params.fundType} অনুদান প্রদান: ৳${params.amount} - ${params.beneficiary}`,
            remarks: params.reason
          },
          ...(db.auditLogs || [])
        ]
      }
    };
  }

  static approveInvestment(
    db: AppDatabaseState,
    params: {
      projectId: string;
      approvedAmount: number;
      approvedBy: string;
      remarks?: string;
    }
  ): { success: boolean; message: string; updatedDb?: AppDatabaseState } {
    const invIndex = (db.investments || []).findIndex(i => i.investmentId === params.projectId);
    if (invIndex === -1) return { success: false, message: 'Investment not found' };
    
    const inv = db.investments[invIndex];
    if (inv.status === 'APPROVED') return { success: false, message: 'This investment is already approved.' };
    if (inv.status !== 'PENDING_APPROVAL' && inv.status !== ('PROPOSED' as any)) return { success: false, message: 'Investment must be pending approval.' };
    if (params.approvedAmount <= 0) return { success: false, message: 'Approved amount must be greater than 0.' };

    const updatedInv = {
      ...inv,
      status: 'APPROVED' as any,
      investmentAmount: params.approvedAmount,
      originalPrincipal: params.approvedAmount,
      outstandingPrincipal: params.approvedAmount,
      expectedReturn: params.approvedAmount + (params.approvedAmount * inv.roiPercentage) / 100,
      approvedAmount: params.approvedAmount,
      approvedBy: params.approvedBy,
      approvalDate: new Date().toISOString(),
      approvalRemarks: params.remarks
    };

    const newInvestments = [...db.investments];
    newInvestments[invIndex] = updatedInv;

    return {
      success: true,
      message: 'বিনিয়োগ সফলভাবে অনুমোদন করা হয়েছে।',
      updatedDb: {
        ...db,
        investments: newInvestments,
        auditLogs: [
          {
            id: Date.now().toString(),
            date: new Date().toISOString(),
            action: 'INVESTMENT_APPROVED',
            performedBy: params.approvedBy,
            details: `Approved investment ${params.projectId} for ${params.approvedAmount}`,
            ipAddress: '127.0.0.1',
          },
          ...(db.auditLogs || [])
        ]
      }
    };
  }

  // Reverse Transaction Safely
  static reverseCollection(
    db: AppDatabaseState,
    receiptNo: string,
    reason: string,
    userName: string
  ): { success: boolean; message: string; updatedDb?: AppDatabaseState } {
    const matchingColls = (db.collections || []).filter(c => c.receiptNo === receiptNo && c.status === 'ACTIVE');
    if (matchingColls.length === 0) return { success: false, message: 'সক্রিয় রসিদ পাওয়া যায়নি!' };

    const coll = matchingColls[0];
    const totalPaidAmount = matchingColls.reduce((sum, c) => sum + c.paidAmount, 0);
    const dateStr = new Date().toISOString().split('T')[0];

    // Mark collection reversed
    const updatedCollections = (db.collections || []).map(c =>
      c.receiptNo === receiptNo ? { ...c, status: 'REVERSED' as const, reversedReason: reason } : c
    );

    // Add reversing ledger entry
    const currentBalance = this.getMemberRunningBalance(db.memberLedgers, coll.memberId);
    const ledgerEntry: MemberLedgerEntry = {
      ledgerId: `LED-${Date.now()}`,
      memberId: coll.memberId,
      date: dateStr,
      voucherNo: `REV-${receiptNo}`,
      receiptNo,
      description: `চাঁদা রসিদ রিভার্সাল (${receiptNo}${matchingColls.length > 1 ? ` - ${matchingColls.length} মাস` : ''}): ${reason}`,
      transactionType: 'REVERSAL',
      debit: totalPaidAmount,
      credit: 0,
      balance: currentBalance - totalPaidAmount,
      reference: receiptNo,
      sourceType: 'REVERSAL',
